name: Publish photo archive
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: false
jobs:
  publish:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Check out the album
        uses: actions/checkout@v4
      - name: Configure Pages
        uses: actions/configure-pages@v5
      - name: Prepare public files
        shell: bash
        run: |
          mkdir -p _site
          cp index.html edit.html README.md .nojekyll _site/
          cp -R assets data photos _site/
      - name: Upload the site
        uses: actions/upload-pages-artifact@v3
        with:
          path: _site
      - name: Publish
        id: deployment
        uses: actions/deploy-pages@v4
